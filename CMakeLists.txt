cmake_minimum_required(VERSION 3.20)
project(FFPR_ScreenReader LANGUAGES C CXX RC)

# One native DLL for FINAL FANTASY I-VI Pixel Remaster, named hid.dll: UnityPlayer.dll imports
# hid.dll statically and hid.dll is not a KnownDLL, so the game loads this copy from its own
# folder. MelonLoader (version.dll) and BepInEx (winhttp.dll) can stay installed: the built-in
# MinHook stacks safely with their hooks. See UNIFIED_MOD_PLAN.md and OPEN_ISSUES.md.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")   # static CRT: no redistributable needed

set(FFPR_DEFS WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
# /EHa: catch (...) in hooks::guarded and ref::call also takes access violations, and
# destructors run while unwinding them. /GR-: no RTTI. /utf-8: sources and literals are UTF-8.
set(FFPR_OPTS /W4 /permissive- /GR- /EHa /utf-8 /MP /Gy /wd4100 /wd4505)

set(FFPR_INCLUDES
    "${CMAKE_SOURCE_DIR}/third_party/sdl3/include"
    "${CMAKE_SOURCE_DIR}/third_party/minhook/include"
    "${CMAKE_SOURCE_DIR}/third_party/tolk")

# --- MinHook 1.3.4, built in from source with the [FFPR] coexistence changes (see
# src/core/minhook_dyn.h): trampolines resume past the first 16 bytes of a target and never
# unhook over another mod's patch, so MelonLoader and BepInEx can hook the same functions.
# Its own warning level: third-party C, kept as close to upstream as possible.
add_library(ffpr_minhook OBJECT
    third_party/minhook/src/buffer.c
    third_party/minhook/src/hook.c
    third_party/minhook/src/trampoline.c
    third_party/minhook/src/hde/hde64.c)
target_include_directories(ffpr_minhook PRIVATE "${CMAKE_SOURCE_DIR}/third_party/minhook/include")
target_compile_options(ffpr_minhook PRIVATE /W3 /utf-8 /Gy)

# --- core: shared by every game. NO include path to src/games (the build enforces that shared
# code cannot call one game's classes). OBJECT library so file-scope registrations
# (core::Install, ref::*) are never dropped by the linker.
file(GLOB_RECURSE FFPR_CORE_SRC CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/core/*.cpp")
add_library(ffpr_core OBJECT ${FFPR_CORE_SRC})
target_include_directories(ffpr_core PRIVATE "${CMAKE_SOURCE_DIR}/src/core" ${FFPR_INCLUDES})
target_compile_definitions(ffpr_core PRIVATE ${FFPR_DEFS})
target_compile_options(ffpr_core PRIVATE ${FFPR_OPTS})

# --- one OBJECT library per game: may include the core, never another game.
set(FFPR_GAME_OBJECTS)
foreach(g ff1 ff2 ff3 ff4 ff5 ff6)
    file(GLOB_RECURSE GAME_SRC CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/games/${g}/*.cpp")
    add_library(ffpr_${g} OBJECT ${GAME_SRC})
    target_include_directories(ffpr_${g} PRIVATE "${CMAKE_SOURCE_DIR}/src/core" "${CMAKE_SOURCE_DIR}/src/games/${g}" ${FFPR_INCLUDES})
    target_compile_definitions(ffpr_${g} PRIVATE ${FFPR_DEFS})
    target_compile_options(ffpr_${g} PRIVATE ${FFPR_OPTS})
    list(APPEND FFPR_GAME_OBJECTS $<TARGET_OBJECTS:ffpr_${g}>)
endforeach()

# --- embedded data: mod UI text (shared + per-game overrides) and each game's content files.
set(FFPR_DATA_RC "${CMAKE_BINARY_DIR}/data.rc")
set(rc_body "// Generated by CMakeLists.txt - the mod's data files, embedded as RCDATA.\n")
macro(ffpr_embed name path)
    if(EXISTS "${CMAKE_SOURCE_DIR}/${path}")
        file(TO_NATIVE_PATH "${CMAKE_SOURCE_DIR}/${path}" native)
        string(REPLACE "\\" "\\\\" native "${native}")
        string(APPEND rc_body "${name} RCDATA \"${native}\"\n")
    endif()
endmacro()
ffpr_embed(MODTEXT_COMMON data/common/mod_text.json)
foreach(n 1 2 3 4 5 6)
    ffpr_embed(MODTEXT_FF${n} data/ff${n}/mod_text.json)
    ffpr_embed(TRANSLATION_FF${n} data/ff${n}/translation.json)
    ffpr_embed(ENTITYNAMES_FF${n} data/ff${n}/EntityNames.json)
    # The baked maps (tools/bake_maps.py from tools/mapdump): grids and routing objects of
    # every sub-map, for the mod-side router (src/core/field/baked_map.cpp).
    ffpr_embed(MAPS_FF${n} data/ff${n}/maps.bin)
    # Guided puzzle routes, authored per game (src/core/field/puzzle_guide.cpp).
    ffpr_embed(PUZZLES_FF${n} data/ff${n}/puzzles.json)
endforeach()
file(GLOB FFPR_SOUNDS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/data/sounds/*.wav")
foreach(s ${FFPR_SOUNDS})
    get_filename_component(sn "${s}" NAME_WE)
    string(TOUPPER "SOUND_${sn}" sid)
    file(TO_NATIVE_PATH "${s}" native)
    string(REPLACE "\\" "\\\\" native "${native}")
    string(APPEND rc_body "${sid} RCDATA \"${native}\"\n")
endforeach()
file(GENERATE OUTPUT "${FFPR_DATA_RC}" CONTENT "${rc_body}")

add_library(hid SHARED
    src/dllmain.cpp
    src/proxy/hid_proxy.cpp
    src/version.rc
    "${FFPR_DATA_RC}"
    $<TARGET_OBJECTS:ffpr_core>
    $<TARGET_OBJECTS:ffpr_minhook>
    ${FFPR_GAME_OBJECTS}
)
target_include_directories(hid PRIVATE "${CMAKE_SOURCE_DIR}/src" "${CMAKE_SOURCE_DIR}/src/core" ${FFPR_INCLUDES})
target_compile_definitions(hid PRIVATE ${FFPR_DEFS})
target_compile_options(hid PRIVATE ${FFPR_OPTS})
# Export the real hid.dll surface (47 names) so the proxy is a drop-in replacement.
target_link_options(hid PRIVATE "/DEF:${CMAKE_SOURCE_DIR}/src/proxy/hid.def" /OPT:REF /OPT:ICF)
# Deliberately no Tolk/SDL import libraries: each is LoadLibrary'd at runtime, so a missing one
# silences the mod instead of stopping the game from starting. MinHook is compiled in (above).
set_target_properties(hid PROPERTIES
    OUTPUT_NAME hid
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/out"
    RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_SOURCE_DIR}/out"
    ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/lib"
    ARCHIVE_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/lib")

# Offline validation host - see smoketest.bat. Not shipped.
add_executable(smoke_host tools/smoke_host.cpp)
target_compile_definitions(smoke_host PRIVATE ${FFPR_DEFS})
set_target_properties(smoke_host PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/out"
    RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_SOURCE_DIR}/out")

# Offline checks of pure-logic code (no game): tools/*_check.cpp, each with the core sources
# it names in its first line comment "// check-sources: a.cpp b.cpp".
file(GLOB FFPR_CHECKS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/tools/*_check.cpp")
foreach(c ${FFPR_CHECKS})
    get_filename_component(cn "${c}" NAME_WE)
    file(STRINGS "${c}" first LIMIT_COUNT 1)
    set(extra)
    if(first MATCHES "check-sources:(.*)")
        separate_arguments(names UNIX_COMMAND "${CMAKE_MATCH_1}")
        foreach(n ${names})
            if(n STREQUAL "+minhook")   # the built-in MinHook (tools/coexist_check.cpp)
                list(APPEND extra $<TARGET_OBJECTS:ffpr_minhook>)
            else()
                list(APPEND extra "${CMAKE_SOURCE_DIR}/src/core/${n}")
            endif()
        endforeach()
    endif()
    add_executable(${cn} "${c}" ${extra})
    target_include_directories(${cn} PRIVATE "${CMAKE_SOURCE_DIR}/src/core" ${FFPR_INCLUDES})
    target_compile_definitions(${cn} PRIVATE ${FFPR_DEFS} FFPR_OFFLINE_CHECK)
    target_compile_options(${cn} PRIVATE /W4 /EHa /utf-8)
    set_target_properties(${cn} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/out/checks"
        RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_SOURCE_DIR}/out/checks")
endforeach()
